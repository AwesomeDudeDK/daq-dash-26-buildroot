# Based on Buildroot's board/raspberrypi/config_cm4io_64bit.txt
# Selected via BR2_PACKAGE_RPI_FIRMWARE_CONFIG_FILE

start_file=start4.elf
fixup_file=fixup4.dat

kernel=Image
arm_64bit=1

disable_overscan=1
gpu_mem_1024=100

# Enable UART0 for serial console on ttyAMA0
dtoverlay=miniuart-bt

# KMS display + V3D GPU; Okay Engine opens /dev/dri/card1
dtoverlay=vc4-kms-v3d

# spidev for the MCP2515 CAN controller (/dev/spidev0.0)
dtparam=spi=on

# USB 2.0 port acts as a device so the board shows up as USB ethernet
dtoverlay=dwc2,dr_mode=peripheral
